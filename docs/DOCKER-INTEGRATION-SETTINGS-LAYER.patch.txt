+        }

+    }

+            .orEmpty()

+            docker = status.flag("docker") == true,
+            services = list,
+            words = words(status["words"] as? JsonObject),
+            why = status.text("why").orEmpty(),
+            available = true,
+        )
+    }

+    fun body(service: String, action: String): String? {
+        if (service.isBlank() || action !in listOf(START, STOP)) return null
+        if (service.any { !(it.isLetterOrDigit() || it == '-' || it == '_' || it == '.') }) return null
+        return "{\"service\":\"$service\",\"action\":\"$action\"}"
+    }

+        p?.title ?: Docker.TITLE,
+        trailing = { Quiet("Refresh", onClick = { reads += 1 }) },
+    ) {
+        Plate {
+            Text(
+                p?.detail ?: Docker.DETAIL,
+                style = MaterialTheme.typography.bodySmall,
+                color = chrome.textMid,
+            )
+            Gap(6)
+            when {
+                p == null && readError != null -> Text(
+                    "Couldn't read it: ${readError.orEmpty()}",
+                    style = MaterialTheme.typography.bodySmall,
+                    color = chrome.warnInk,
+                )
+                p == null -> Text(
+                    "Reading…",
+                    style = MaterialTheme.typography.bodySmall,
+                    color = chrome.textLo,
+                )
+                p.line.isNotEmpty() -> Text(
+                    p.line,
+                    style = MaterialTheme.typography.bodySmall,
+                    color = chrome.warnInk,
+                )
+                else -> {
+                    p.services.forEach { s ->
+                        Gap(8)
+                        Row(verticalAlignment = Alignment.CenterVertically) {
+                            Text(
+                                s.key.replaceFirstChar { it.uppercase() },
+                                style = MaterialTheme.typography.titleSmall,
+                                color = if (s.running) chrome.okInk else chrome.textHi,
+                                modifier = Modifier.weight(1f),
+                            )
+                            // The state in words, never by colour alone.
+                            Text(
+                                s.stateWord,
+                                style = MaterialTheme.typography.labelSmall,
+                                color = chrome.textMid,
+                            )
+                        }
+                        if (s.provides.isNotEmpty()) {
+                            Text(
+                                s.provides,
+                                style = MaterialTheme.typography.bodySmall,
+                                color = chrome.textMid,
+                            )
+                        }
+                        // Where it answers, so "this PC only" is visible rather
+                        // than merely claimed.
+                        if (s.ours && s.published.isNotEmpty()) {
+                            Text(
+                                "${s.published} - this PC only",
+                                style = MaterialTheme.typography.labelSmall,
+                                color = chrome.textLo,
+                            )
+                        }
+                        Gap(4)
+                        when {
+                            // Not one of Jarvis's: say so, offer nothing. The PC
+                            // would refuse anyway; showing no button is the
+                            // honest version of that.
+                            !s.present || !s.ours -> Text(
+                                s.why.ifEmpty { Docker.NOT_OURS },
+                                style = MaterialTheme.typography.labelSmall,
+                                color = chrome.warnInk,
+                            )
+                            else -> Row {
+                                Quiet(
+                                    "Start",
+                                    // Starting asks on the PC and is held while
+                                    // the link is stale (rule 4), so it needs a
+                                    // fresh link as well as an image already here.
+                                    enabled = !busy && s.canStart && !s.running && canAct,
+                                    onClick = {
+                                        busy = true
+                                        said = null
+                                        go(scope, s.key, Docker.START) { said = it; busy = false; reads += 1 }
+                                    },
+                                )
+                                Gap(10)
+                                Quiet(
+                                    "Stop",
+                                    // Stopping is never held on a stale link: it
+                                    // only makes Jarvis do less.
+                                    enabled = !busy && s.canStop && s.running,
+                                    onClick = {
+                                        busy = true
+                                        said = null
+                                        go(scope, s.key, Docker.STOP) { said = it; busy = false; reads += 1 }
+                                    },
+                                )
+                            }
+                        }
+                        // A start whose image is not here is refused by the PC,
+                        // and this is the PC's own sentence saying why.
+                        if (s.ours && !s.canStart && s.why.isNotEmpty()) {
+                            Text(
+                                s.why,
+                                style = MaterialTheme.typography.labelSmall,
+                                color = chrome.warnInk,
+                            )
+                        }
+                    }
+                    Gap(6)
+                    Text(
+                        if (busy) Docker.ASKING else Docker.OPENS_A_CARD,
+                        style = MaterialTheme.typography.labelSmall,
+                        color = chrome.textLo,
+                    )
+                    if (!canAct) {
+                        Text(
+                            Docker.WAITING_LINK,
+                            style = MaterialTheme.typography.labelSmall,
+                            color = chrome.warnInk,
+                        )
+                    }
+                }
+            }
+            said?.let {
+                Gap(4)
+                Text(
+                    it,
+                    style = MaterialTheme.typography.labelSmall,
+                    color = chrome.textMid,
+                    modifier = Modifier.liveStatus(),
+                )
+            }
+        }
+    }

+/** One start or stop, with the sentence the PC answered shown under the row. */
+private fun go(
+    scope: kotlinx.coroutines.CoroutineScope,
+    service: String,
+    action: String,
+    done: (String) -> Unit,
+) {
+    scope.launch {
+        val said = try {
+            JarvisRuntime.dockerService(service, action)
+        } catch (e: Exception) {
+            "Not changed."
+        }
+        done(said)
+    }
